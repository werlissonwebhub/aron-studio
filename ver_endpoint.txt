358: def _extract_json(text: str) -> str:
359:     if not text:
360:         return ""
361:     t = text.strip()
362:     if t.startswith("```"):
363:         t = t[3:]
364:         if t[:4].lower() == "json":
365:             t = t[4:]
366:         if t.endswith("```"):
367:             t = t[:-3]
368:         t = t.strip()
369:     first_obj = t.find("{")
370:     first_arr = t.find("[")
371:     starts = [p for p in (first_obj, first_arr) if p != -1]
372:     if starts:
373:         start = min(starts)
374:         end = max(t.rfind("}"), t.rfind("]"))
375:         if end > start:
376:             t = t[start:end + 1]
377:     return t.strip()
